# 5. Executing BDs with total net payment < $1,000 per month/year
# 5a. Aggregated across all stock groups
```sql
SELECT 
    executing_bd,
    month,
    year,
    SUM(
        COALESCE(netpmtpaidrecvmarketordersusd, 0) +
        COALESCE(netpmtpaidrecvmarketablelimitordersusd, 0) +
        COALESCE(netpmtpaidrecvnonmarketablelimitordersusd, 0)
    ) AS total_net_payment_usd
FROM executing_bd_606
WHERE executing_bd IS NOT NULL
  AND month IS NOT NULL
  AND year IS NOT NULL
GROUP BY executing_bd, month, year
HAVING SUM(
        COALESCE(netpmtpaidrecvmarketordersusd, 0) +
        COALESCE(netpmtpaidrecvmarketablelimitordersusd, 0) +
        COALESCE(netpmtpaidrecvnonmarketablelimitordersusd, 0)
    ) < 1000
ORDER BY year DESC, month, executing_bd;


```
# 5b. Aggregated by executing_bd + month + year + stock_group
```sql
SELECT 
    executing_bd,
    month,
    year,
    stock_group,
    SUM(
        COALESCE(netpmtpaidrecvmarketordersusd, 0) +
        COALESCE(netpmtpaidrecvmarketablelimitordersusd, 0) +
        COALESCE(netpmtpaidrecvnonmarketablelimitordersusd, 0)
    ) AS total_net_payment_usd
FROM executing_bd_606
WHERE executing_bd IS NOT NULL
  AND month IS NOT NULL
  AND year IS NOT NULL
  AND stock_group IS NOT NULL
GROUP BY executing_bd, month, year, stock_group
HAVING SUM(
        COALESCE(netpmtpaidrecvmarketordersusd, 0) +
        COALESCE(netpmtpaidrecvmarketablelimitordersusd, 0) +
        COALESCE(netpmtpaidrecvnonmarketablelimitordersusd, 0)
    ) < 1000
ORDER BY year DESC, month, executing_bd, stock_group;
```
